#include <iostream>

using namespace std;

int main()

{

    int a[10], n, key;

    int i;

    cout << "Enter number of
elements: ";

    cin >> n;

    cout << "Enter elements in
sorted order: ";

    for(i = 0; i < n; i++)

    {

        cin >> a[i];

    }

    cout << "Enter element to
search: ";

    cin >> key;

    // Linear Search

    cout << "\nLinear Search:";

    for(i = 0; i < n; i++)

    {

        if(a[i] == key)

        {

            cout << "\nElement found
at position " << i + 1;

            break;

        }

    }

    if(i == n)

    {

        cout << "\nElement not
found";

    }

    // Binary Search

    int low = 0, high = n - 1, mid;

    bool found = false;

    while(low <= high)

    {

        mid = (low + high) / 2;

        if(a[mid] == key)

        {

            found = true;

            break;

        }

        else if(key < a[mid])

        {

            high = mid - 1;

        }

        else

        {

            low = mid + 1;

        }

    }

    cout << "\n\nBinary
Search:";

    if(found)

        cout << "\nElement found at
position " << mid + 1;

    else

        cout << "\nElement not
found";

    return 0;

}

Program 7 – A) Infix to Postfix Expression

#include <iostream>

#include <stack>

using namespace std;

// Function to check operator
priority

int priority(char ch) //This
function finds the priority of an operator

{

    if(ch == '+' || ch == '-') //

        return 1;

    if(ch == '*' || ch == '/')

        return 2;

    return 0;

}

int main()

{

    string infix, postfix = "";
 //Creates two string variables

    stack<char> s;                                                                           //Creates
a stack named s

    cout << "Enter infix expression:
";

    cin >> infix;

    for(int i = 0; i < infix.length(); i++)                                   //This loop
reads every character of the infix expression

    {

        char ch = infix[i];

        // If operand, add directly to postfix

        if((ch >= 'A' && ch <=
'Z') ||                                                //This checks whether ch is an operand.

           (ch >= 'a' && ch <=
'z') ||

           (ch >= '0' && ch <=
'9'))

        {

            postfix = postfix + ch;

        }







                                                                               // If opening bracket

        else if(ch == '(')                                              //Checks
whether the character is an opening bracket.

        {

            s.push(ch);
 //If it is (, push it into the
stack

        }

                                                                             // If closing bracket

        else if(ch == ')')                                              //Checks
whether the character is a closing bracket.

        {

            while(!s.empty() && s.top()
!= '(')                           //Continue removing
elements from the stack until:(

            {

                postfix = postfix +
s.top();                                     //Take the
top operator and add it to postfix.

                s.pop();                                                                        //Remove
the top element from the stack.

            }

            if(!s.empty())                                                 //After all operators are removed, remove the
opening bracket (.

                s.pop();

        }

                                                                         //
If operator

        else

        {

            while(!s.empty() &&

                  priority(s.top()) >=
priority(ch))

            {

                postfix = postfix + s.top();               //If the stack operator has higher or equal
priority, add it to postfix.

                s.pop();
  //Remove that operator from the
stack.

            }

            s.push(ch);
  //After removing higher/equal
priority operators,

                                                                                                           // push the current operator into the
stack.

        }

    }





    // Remove remaining operators

    while(!s.empty())   //checks whether operators are still present
in the stack

    {

        postfix = postfix + s.top(); //Adds the
top operator to postfix.

        s.pop();    //Removes that operator.

    }

    cout << "Postfix expression =
" << postfix;

     //Displays the final postfix expression.

    return 0;

}

#include <iostream>

{

    stack<int> s;

    string exp;

    int a, b;

    cout << "Enter postfix
expression: ";

    cin >> exp;

    for(int i = 0; i < exp.length(); i++)

    {

        // If it is an operand

        if(exp[i] >= '0' && exp[i]
<= '9')

        {

            s.push(exp[i] - '0');

        }





        // If it is an operator

        else

        {

            a = s.top();

            s.pop();

            b = s.top();

            s.pop();

            if(exp[i] == '+')

                s.push(b + a);

            else if(exp[i] == '-')

                s.push(b - a);

            else if(exp[i] == '*')

                s.push(b * a);

            else if(exp[i] == '/')

                s.push(b / a);

        }

    }
